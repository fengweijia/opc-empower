  let totalSteps = userFlowSteps.length
  
  for (const step of userFlowSteps) {

    console.log(`   操作: ${step.action}`)

      console.log(`   详情: ${step.details}`)

      console.log(`   ✅ 通过\n`)
      passedSteps++

      console.log(`   ❌ 失败 - 需要检查\n`)
    }
  }
  
  // 测试总结
  console.log('📈 用户流程测试总结')
  console.log('='.repeat(50))
  console.log(`总步骤数: ${totalSteps}`)
  console.log(`通过步骤: ${passedSteps}`)
  console.log(`失败步骤: ${totalSteps - passedSteps}`)
  console.log(`通过率: ${((passedSteps / totalSteps) * 100).toFixed(1)}%`)
  console.log()
  
  if (passedSteps === totalSteps) {
    console.log('🎉 所有用户流程测试通过！')
    console.log('💡 用户可以从配置到使用完整流程无阻碍')
  } else if (passedSteps >= totalSteps * 0.8) {

    console.log('💡 建议重点检查失败步骤的用户体验')

    console.log('❌ 流程测试发现问题较多')
    console.log('💡 需要重新设计关键路径的用户体验')
